	\tiny \textbf{ERA5 Heat Flux Sign Convention:} \\
	\tiny
	\begin{itemize}
		\item Surface Latent Heat Flux: $+$ = condensation (heat to surface), $-$ = evaporation (heat to atmosphere) \\
		\item Surface Sensible Heat Flux: $+$ = warm atmosphere $\rightarrow$ cool surface, $-$ = warm surface $\rightarrow$ cool atmosphere \\
		\item Total Heat Flux: $+$ = net heat gain to surface, $-$ = net heat loss from surface (surface cooling)
	\end{itemize}

		\column{0.2\textwidth}
		\vspace{1cm}
		\begin{align*}\footnotesize
			\zeta_{cart, rel} & = \frac{\partial v}{\partial x} - \frac{\partial u}{\partial y} \\
			\zeta_{cart, abs} & = \zeta_{rel} + f                                               \\
			\text{where } f   & = 2\Omega\sin\phi
		\end{align*}

		\vspace{1.5cm}
		\begin{align*}\footnotesize
			\zeta_{polar} & = \frac{V_t}{r} + \frac{\partial V_t}{\partial r}
		\end{align*}

		\vspace{0.1cm}
		\footnotesize \textbf{NaN from 0-25km:}
		\begin{itemize}
			\item At $r=0$, $\frac{V_t}{r}$ is undefined (division by zero)
		\end{itemize}
