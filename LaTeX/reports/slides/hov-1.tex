			\begin{itemize}\tiny
				\item Relative Vorticity :
				      \begin{align*}
					      \zeta_{cart, rel} & = \frac{\partial v}{\partial x} - \frac{\partial u}{\partial y} \\
				      \end{align*}
				\item Absolute Vorticity :
				      \begin{align*}
					      \zeta_{cart, abs} & = \zeta_{rel} + f \\
					      \text{where } f   & = 2\Omega\sin\phi
				      \end{align*}
			\end{itemize}

			\begin{itemize}\tiny
				\item Saturation vapor pressure :
				      \begin{align*}
					      e_s = 6.112 \exp\left(\frac{17.67 * d2m}{d2m + 243.5}\right)
				      \end{align*}
				      where \texttt{d2m} = 2m dew point in °C
				\item Mixing ratio :
				      \begin{align*}
					      w = 0.622 \frac{e_s}{sp - e_s} \times 1000
				      \end{align*}
				      where \texttt{sp} = surface pressure (hPa)
			\end{itemize}
